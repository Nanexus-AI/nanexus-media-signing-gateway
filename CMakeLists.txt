cmake_minimum_required(VERSION 3.18)
project(nanexus_gateway VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(gateway_core src/config.cpp src/media.cpp)
target_include_directories(gateway_core PUBLIC include)
target_compile_options(gateway_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(nanexus-gateway src/main.cpp)
target_link_libraries(nanexus-gateway PRIVATE gateway_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(gateway-tests tests/contracts.cpp)
  target_link_libraries(gateway-tests PRIVATE gateway_core)
  add_test(NAME contracts COMMAND gateway-tests)
  add_test(NAME config-example COMMAND nanexus-gateway --check-config ${CMAKE_CURRENT_SOURCE_DIR}/config/gateway.example.conf)
endif()

option(NANEXUS_NATIVE "Build pinned native media runtime" OFF)
if(NANEXUS_NATIVE)
  include(cmake/native.cmake)
endif()
